<<!DOCTYPE html>
<html>
<head>
  <title>My Web from Google Sheet</title>
</head>
<body>
  <h1>ข้อมูลจาก Google Sheets</h1>
  <div id="sheet-data"></div>

  <script>
    const sheetID = "2PACX-1vQGQVGiyX1bwknbAhOtaV--D17wq8waiFyyDOUPYLE0pdBzhUXF6-FChvNysfBZcZwMJ92ANXr_x26n";
    const sheetName = "Sheet1";
    const url = `https://docs.google.com/spreadsheets/d/${sheetID}/gviz/tq?sheet=${sheetName}`;

    fetch(url)
      .then(res => res.text())
      .then(rep => {
        const data = JSON.parse(rep.substring(47).slice(0, -2));
        let html = "<table border='1'><tr>";
        const cols = data.table.cols.map(col => `<th>${col.label}</th>`).join('');
        html += cols + "</tr>";

        data.table.rows.forEach(row => {
          html += "<tr>";
          row.c.forEach(cell => {
            html += `<td>${cell ? cell.v : ""}</td>`;
          });
          html += "</tr>";
        });

        document.getElementById("sheet-data").innerHTML = html;
      });
  </script>
</body>
</html>
